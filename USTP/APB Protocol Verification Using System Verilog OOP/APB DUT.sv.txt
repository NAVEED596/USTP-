module apb_slave ( 
input  logic        pclk, 
input  logic        presetn, 
input  logic [31:0] paddr, 
input  logic        psel, 
input  logic        penable, 
input  logic        pwrite, 
input  logic [31:0] pwdata, 
output logic [31:0] prdata, 
output logic        pready, 
output logic        pslverr 
); 
logic [31:0] reg_0x00; 
logic [31:0] reg_0x04; 
typedef enum logic [1:0] { 
IDLE   = 2'b00, 
SETUP  = 2'b01, 
ACCESS = 2'b10 
} state_t; 
state_t current_state, next_state; 
always_ff @(posedge pclk or negedge presetn) begin 
if (!presetn) 
current_state <= IDLE; 
else 
current_state <= next_state; 
end 
always_comb begin 
case (current_state) 
IDLE: begin 
if (psel && !penable) 
next_state = SETUP; 
else 
next_state = IDLE; 
end 
SETUP: begin 
                if (psel && penable) 
                    next_state = ACCESS; 
                else 
                    next_state = IDLE; 
            end 
            ACCESS: begin 
                if (pready) begin 
                    if (psel && !penable) 
                        next_state = SETUP; 
                    else 
                        next_state = IDLE; 
                end else begin 
                    next_state = ACCESS; 
                end 
            end 
            default: next_state = IDLE; 
        endcase 
    end 
 
    always_ff @(posedge pclk or negedge presetn) begin 
        if (!presetn) begin 
            prdata   <= 32'h0; 
            pready   <= 1'b1; 
            pslverr  <= 1'b0; 
            reg_0x00 <= 32'h0; 
            reg_0x04 <= 32'h0; 
        end else begin 
            pready  <= 1'b1; 
            pslverr <= 1'b0; 
 
            if (psel && penable) begin 
                if (pwrite) begin 
                    case (paddr) 
                        32'h00000000: reg_0x00 <= pwdata; 
                        32'h00000004: reg_0x04 <= pwdata; 
                        default:     pslverr  <= 1'b1; 
                    endcase 
                end else begin 
                    case (paddr) 
                        32'h00000000: prdata <= reg_0x00; 
                        32'h00000004: prdata <= reg_0x04; 
                        default: begin 
                            prdata  <= 32'h0; 
                            pslverr <= 1'b1; 
                        end 
                    endcase 
                end 
            end 
        end 
    end 
 
endmodule

`timescale 1ns / 1ps
module tb_vending_machine;
    reg clk;
    reg rst_n;
    reg nickel;
    reg dime;
    wire dispense;
    wire change;
    vending_machine uut (
        .clk(clk), 
        .rst_n(rst_n), 
        .nickel(nickel), 
        .dime(dime), 
        .dispense(dispense), 
        .change(change)
    );
    always #10 clk = ~clk;
    initial begin
        clk = 0;
        rst_n = 0;
        nickel = 0;
        dime = 0;
        #20;
        rst_n = 1;
        @(posedge clk); nickel = 1; dime = 0;
        @(posedge clk); nickel = 1; dime = 0;
        @(posedge clk); nickel = 1; dime = 0;
        @(posedge clk); nickel = 0; dime = 0;
        #40;
        @(posedge clk); nickel = 1; dime = 0;
        @(posedge clk); nickel = 0; dime = 1;
        @(posedge clk); nickel = 0; dime = 0;
        #20;
        @(posedge clk); nickel = 0; dime = 1;
        @(posedge clk); nickel = 0; dime = 1;
        @(posedge clk); nickel = 0; dime = 0;
        #40;
        $finish;
    end
endmodule

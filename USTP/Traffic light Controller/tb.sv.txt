`timescale 1ns / 1ps
 
module tb_traffic_light_controller;
 
    reg clk;
    reg rst_n;
 
    wire [1:0] light_out;
 
    traffic_light_controller uut (
        .clk(clk), 
        .rst_n(rst_n), 
        .light_out(light_out)
    );
 
    always #10 clk = ~clk;
 
    initial begin
        clk = 0;
        rst_n = 0;
 
        #20;
        rst_n = 1;
        
        #120;
        
        $finish;
    end
      
endmodule

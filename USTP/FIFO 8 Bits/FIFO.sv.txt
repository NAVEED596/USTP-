module synchronous_fifo_8bit (
    input wire clk,
    input wire rst_n,
    input wire wr_en,
    input wire rd_en,
    input wire [7:0] din,
    output reg [7:0] dout,
    output reg full,
    output reg empty
);
    reg [7:0] mem [0:7];
    reg [2:0] wr_ptr;
    reg [2:0] rd_ptr;
    reg [3:0] count;
    always @(*) begin
        empty = (count == 4'd0);
        full  = (count == 4'd8);
    end
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            wr_ptr <= 3'd0;
            rd_ptr <= 3'd0;
            count  <= 4'd0;
            dout   <= 8'd0;
            mem[0] <= 8'd0;
            mem[1] <= 8'd0;
            mem[2] <= 8'd0;
            mem[3] <= 8'd0;
            mem[4] <= 8'd0;
            mem[5] <= 8'd0;
            mem[6] <= 8'd0;
            mem[7] <= 8'd0;
        end else begin
            if (wr_en && !full && !(rd_en && !empty)) begin
                mem[wr_ptr] <= din;
                wr_ptr      <= wr_ptr + 1;
                count       <= count + 1;
            end
            else if (rd_en && !empty && !(wr_en && !full)) begin
                dout   <= mem[rd_ptr];
                rd_ptr <= rd_ptr + 1;
                count  <= count - 1;
            end
            else if (wr_en && !full && rd_en && !empty) begin
                mem[wr_ptr] <= din;
                wr_ptr      <= wr_ptr + 1;
                dout        <= mem[rd_ptr];
                rd_ptr      <= rd_ptr + 1;
            end
        end
    end
endmodule

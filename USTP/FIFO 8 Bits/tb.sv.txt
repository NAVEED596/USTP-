`timescale 1ns / 1ps
module tb_synchronous_fifo_8bit;
    reg clk;
    reg rst_n;
    reg wr_en;
    reg rd_en;
    reg [7:0] din;
    wire [7:0] dout;
    wire full;
    wire empty;
    synchronous_fifo_8bit uut (
        .clk(clk),
        .rst_n(rst_n),
        .wr_en(wr_en),
        .rd_en(rd_en),
        .din(din),
        .dout(dout),
        .full(full),
        .empty(empty)
    );
    always #10 clk = ~clk;
    initial begin
        clk = 0;
        rst_n = 0;
        wr_en = 0;
        rd_en = 0;
        din = 0;
        #20;
        rst_n = 1;
        #20;
        @(posedge clk); wr_en = 1; din = 8'h11;
        @(posedge clk); wr_en = 1; din = 8'h22;
        @(posedge clk); wr_en = 1; din = 8'h33;
        @(posedge clk); wr_en = 1; din = 8'h44;
        @(posedge clk); wr_en = 1; din = 8'h55;
        @(posedge clk); wr_en = 1; din = 8'h66;
        @(posedge clk); wr_en = 1; din = 8'h77;
        @(posedge clk); wr_en = 1; din = 8'h88;
        @(posedge clk); wr_en = 0;
        #20;
        @(posedge clk); wr_en = 1; din = 8'hFF;
        @(posedge clk); wr_en = 0;
        #20;
        @(posedge clk); rd_en = 1;
        @(posedge clk); rd_en = 1;
        @(posedge clk); rd_en = 1;
        @(posedge clk); rd_en = 1;
        @(posedge clk); rd_en = 1;
        @(posedge clk); rd_en = 1;
        @(posedge clk); rd_en = 1;
        @(posedge clk); rd_en = 1;
        @(posedge clk); rd_en = 0;
        #20;
        @(posedge clk); rd_en = 1;
        @(posedge clk); rd_en = 0;
        #20;
        @(posedge clk); wr_en = 1; rd_en = 1; din = 8'h99;
        @(posedge clk); wr_en = 0; rd_en = 0;
        #40;
        $finish;
    end
endmodule

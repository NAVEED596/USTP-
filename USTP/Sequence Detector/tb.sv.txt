`timescale 1ns / 1ps

module tb_sequence_detector_1101;

    // Inputs
    reg clk;
    reg rst_n;
    reg sequence_in;

    // Outputs
    wire detector_out;

    // Instantiate the Unit Under Test (UUT)
    sequence_detector_1101 uut (
        .clk(clk), 
        .rst_n(rst_n), 
        .sequence_in(sequence_in), 
        .detector_out(detector_out)
    );

    // Clock generation (50MHz)
    always #10 clk = ~clk;

    initial begin
        // Initialize Inputs
        clk = 0;
        rst_n = 0;
        sequence_in = 0;

        // Hold reset state for 20ns
        #20;
        rst_n = 1;
        
        // Feed sequence: 1 -> 1 -> 0 -> 1 (Should detect)
        @(posedge clk); sequence_in = 1;
        @(posedge clk); sequence_in = 1;
        @(posedge clk); sequence_in = 0;
        @(posedge clk); sequence_in = 1; // Reaches S_1101 on next clock edge
        
        // Let it cycle one more time to observe the Moore output peak
        @(posedge clk); sequence_in = 0; 
        
        // Feed partial sequence and break it: 1 -> 1 -> 1 -> 0 -> 1
        @(posedge clk); sequence_in = 1;
        @(posedge clk); sequence_in = 1;
        @(posedge clk); sequence_in = 1; 
        @(posedge clk); sequence_in = 0;
        @(posedge clk); sequence_in = 1;

        // Finish simulation
        @(posedge clk);
        #40;
        $finish;
    end
      
endmodule
